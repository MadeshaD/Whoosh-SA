cmake_minimum_required(VERSION 3.16)
project(WhooshSA LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
    message(FATAL_ERROR "Configure for 32-bit Windows (Visual Studio: -A Win32).")
endif()
add_library(Whoosh MODULE source/plugin.cpp source/camera.cpp source/camera_game.cpp source/visuals.cpp source/detector.cpp source/audio.cpp source/mixer.cpp)
add_custom_command(TARGET Whoosh POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
    "${CMAKE_CURRENT_SOURCE_DIR}/Whoosh.txd" "$<TARGET_FILE_DIR:Whoosh>/Whoosh.txd")
target_compile_features(Whoosh PRIVATE cxx_std_17)
target_link_libraries(Whoosh PRIVATE winmm user32 ole32)
set_target_properties(Whoosh PROPERTIES PREFIX "" OUTPUT_NAME "Whoosh.SA" SUFFIX ".asi")
if(MSVC)
    set_property(TARGET Whoosh PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
elseif(MINGW)
    target_compile_options(Whoosh PRIVATE $<$<COMPILE_LANGUAGE:CXX>:-mstackrealign>)
    target_link_options(Whoosh PRIVATE -static -Wl,--exclude-all-symbols)
endif()
